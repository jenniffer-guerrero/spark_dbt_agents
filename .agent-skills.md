  incremental_mapping:
    description: >
      Delta-driven recomputation of the REL_ID -> REL_ID_REGIE_KLANT mapping.
      Only nodes affected by source changes are reprocessed.
    source_tables:
      - shebang.direct_bank: edges rel_id <-> np_sbl_id
      - shebang.ggm_np:      edges rel_id <-> ikb_no
    state_tables:
      - edge_snapshot:    columns [source_table, rel_id, id_key]  — previous run edges
      - seed_snapshot:    columns [seed_rel_id]                   — previous run seeds
      - seed_membership:  columns [rel_id, rel_id_regie_klant]    — previous run mapping
    output_table:
      shebang.output: columns [REL_ID, REL_ID_REGIE_KLANT]
    seed_rule: >
      Active REGIE seeds are direct_bank rows where drc_bnk_f = 'Y'
      and edl_valid_to_dts starts with '9999-12-31'.
    winner_rule: >
      When multiple active seeds reach the same rel_id, assign the minimum
      numeric seed (cast to bigint). If no numeric seed exists, use the
      lexical minimum.
    phases:
      1_build_current_snapshot: >
        Read source tables. Build normalized edge list (cast keys to string,
        drop nulls and blank values, deduplicate). Build active seed set.
      2_detect_deltas: >
        Compare current edges with edge_snapshot to find edge_added and
        edge_removed. Compare current seeds with seed_snapshot to find
        seed_added and seed_removed. Find rel_ids previously mapped to
        removed seeds via seed_membership.
      3_compute_impact: >
        Union changed rel_ids, added/removed seeds, and rows linked to removed
        seeds into an impact start set. Expand using BFS (expand_rel_component)
        over the full edge graph. On first run (state tables empty), treat all
        rel_ids as impacted.
      4_recompute_impacted: >
        For the impacted active seeds only, expand (seed_rel_id, rel_id)
        reachability pairs using BFS (expand_seed_pairs). Apply winner_rule
        to assign REL_ID_REGIE_KLANT per rel_id.
      5_merge_and_enforce: >
        Keep unimpacted rows from previous output. Replace impacted rows with
        recomputed values. Retain only rows whose REL_ID_REGIE_KLANT is still
        an active seed.
      6_persist: >
        Write output and all three state tables using the staging-table swap
        pattern (see spark_runtime_reliability). Clear cache after all writes.
    bfs_parameters:
      max_iter: 40
      checkpoint_every: 2
    observability_dict:
      changed_edges: int
      added_seeds: int
      removed_seeds: int
      impacted_rel_ids: int
      output_rows: int

  spark_runtime_reliability:
    description: >
      Reliable Spark writes and lineage management for local Parquet/Hive
      environments where UPDATE and MERGE are not robustly supported.
    staging_swap_pattern: |
      # Atomically replace a Parquet table via staging rename:
      spark.sql(f"DROP TABLE IF EXISTS {table}__staging")
      df.write.mode('overwrite').format('parquet').saveAsTable(f"{table}__staging")
      spark.sql(f"DROP TABLE IF EXISTS {table}")
      spark.sql(f"ALTER TABLE {table}__staging RENAME TO {table}")
      spark.catalog.refreshTable(table)
    checkpointing: |
      # Break long BFS lineage chains:
      df = df.checkpoint(eager=True)   # call every checkpoint_every steps
    cache_invalidation: |
      spark.catalog.clearCache()       # call after all writes in a run
    empty_check: |
      # Cheap emptiness test — avoids a full count job:
      def _has_rows(df) -> bool:
          return df.limit(1).count() > 0
    local_constraints:
      - Wrap CREATE DATABASE in try/except; some restricted catalogs reject it.
      - Wrap setCheckpointDir in try/except; fall back to checkpoint_every=0.

  stateful_test_design:

      Each case mutates sources, runs the incremental update, and verifies

        'expected_removed':   list[tuple[str, str]],
        'actual_inserted':    list[tuple[str, str]],
        'actual_removed':     list[tuple[str, str]],
        'failure_class':      'logic_defect' | 'data_precondition' | 'environment_runtime' | 'none',
        'suggested_fix':      str,    # name of the function to repair, or 'none'
      }
    required_helpers:
      pair_set: "pair_set(df) -> set[tuple[str,str]] — collect (REL_ID, REL_ID_REGIE_KLANT)"
      assert_delta: "assert_delta(prev, cur, exp_ins, exp_rem) — strict set diff; print diff on failure"
      assert_no_duplicates: "assert_output_has_no_duplicates(df) — verify REL_ID is unique"
      make_feedback: "make_feedback(case, status, exp_ins, exp_rem, act_ins, act_rem, failure_class, fix) -> dict"

    per_case_sequence: |

      2.  Apply source mutation via staging-swap (spark_runtime_reliability)

      5.  feedback = make_feedback(...)
      6.  If feedback['failure_class'] == 'logic_defect': invoke agent_repair_protocol
      7.  assert_delta(prev, cur, expected_inserted, expected_removed)
      8.  assert_output_has_no_duplicates(current output)
      9.  Print pass / fail / unresolved for this case
      10. last_feedback = feedback
    baseline_management: |

  agent_repair_protocol:
    description: >
      Closed-loop repair of logic defects in siebel_id_update.ipynb.
      Triggered by the test notebook when failure_class == 'logic_defect'.
      The test notebook calls the data-engineer-analytics agent via the
      Anthropic API to diagnose and patch the failing function.
    trigger: "feedback['failure_class'] == 'logic_defect'"
    max_retries: 2
    escalation: >
      After 2 failed repair attempts, set feedback['status'] = 'unresolved'
      and continue to the next case without further repair attempts.
    call_repair_agent: |
      def call_repair_agent(feedback: dict, update_nb_path: str) -> bool:
          """
          Invoke the data-engineer-analytics agent to patch the update notebook.
          Returns True if the notebook was patched, False on any error.
          """
          import json, os, anthropic

          # Read agent instructions and notebook content.
          with open('.github/agents/data-engineer-analytics.agent.md') as f:
              agent_instructions = f.read()
          with open(update_nb_path) as f:
              nb = json.load(f)
          code_cells = [(''.join(c['source'])) for c in nb['cells'] if c['cell_type'] == 'code']

          model = os.environ.get('ANTHROPIC_MODEL', 'claude-opus-4-7')
          client = anthropic.Anthropic()
          response = client.messages.create(
              model=model,
              max_tokens=4096,
              system=agent_instructions,
              messages=[{
                  'role': 'user',
                  'content': (
                      f"A test case failed.\n\nFeedback:\n{json.dumps(feedback, indent=2)}\n\n"
                      f"Update notebook code cells:\n\n"
                      + '\n\n---\n\n'.join(code_cells)
                      + f"\n\nFix the logic defect. Return ONLY the corrected Python source "
                        f"for the function '{feedback['suggested_fix']}'. No explanation."
                  )
              }]
          )
          corrected_source = response.content[0].text.strip()

          # Locate and replace the target cell.
          target = f"def {feedback['suggested_fix']}("
          for cell in nb['cells']:
              if cell['cell_type'] == 'code' and target in ''.join(cell['source']):
                  cell['source'] = corrected_source
                  with open(update_nb_path, 'w') as f:
                      json.dump(nb, f, indent=1)
                  return True
          return False
    retry_sequence: |
      for attempt in range(1, agent_repair_protocol.max_retries + 1):
          print(f"Repair attempt {attempt} for case '{feedback['case']}'...")
          patched = call_repair_agent(feedback, update_nb_path)
          if patched:
              reload_update_logic(globals())
              # Re-run steps 3–5 of per_case_sequence
              stats = run_incremental_update(spark, test_feedback=feedback)
              cur = pair_set(current output)
              feedback = make_feedback(...)
              if feedback['status'] == 'pass':
                  break
      else:
          feedback['status'] = 'unresolved'
          print(f"UNRESOLVED after {agent_repair_protocol.max_retries} attempts.")
